/**
 * Runs pending upgrades once per site, when the stored version is older than the code.
 */
function productreels_maybe_upgrade() {
	$stored = get_option( 'productreels_version', '1.0.0' ); // Use the existing option if 1.0.0 already stores one.

	if ( version_compare( $stored, PRODUCTREELS_VERSION, '>=' ) ) {
		return;
	}

	if ( version_compare( $stored, '1.1.0', '<' ) ) {
		productreels_upgrade_to_110();
	}

	update_option( 'productreels_version', PRODUCTREELS_VERSION, false );
}
add_action( 'plugins_loaded', 'productreels_maybe_upgrade', 20 );
